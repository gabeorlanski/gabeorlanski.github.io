```python title="Opus 5.5 · pyda (LDB library) · attempt 2 · src/pyda/io.py:545-557" startLineNumber=545 {12,13}
def _matching_row_groups(f: FileInfo, arrow_filter) -> list[int] | None:
    """Row groups whose statistics may satisfy ``arrow_filter`` (None = all)."""
    if arrow_filter is None:
        return None
    try:
        if f.fs is not None:
            dset = pds.dataset(f.path, format="parquet", filesystem=f.fs)
        else:
            dset = pds.dataset(f.path, format="parquet")
        frag = next(iter(dset.get_fragments()))
        return [rg.id for rg in frag.subset(filter=arrow_filter).row_groups]
    except Exception:  # statistics unusable (e.g. type mismatch) -> read everything
        return None
```
