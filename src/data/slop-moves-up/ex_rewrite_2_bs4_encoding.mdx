def find_declared_encoding(data):
    window = data[:2048]
    m = re.search(
        rb"<\?xml(?:\s|\?)[^>]*\bencoding\s*=\s*[\"']([^\"']+)[\"']",
        window,
        re.I,
    )
    if m:
        return m.group(1).decode("ascii", "replace").strip()
    for mt in re.finditer(rb"<meta\b[^>]*>", window, re.I):
        tag = mt.group(0)
        m = re.search(rb"charset\s*=\s*[\"']?\s*([^\"'\s;>]+)", tag, re.I)
        if m:
            return m.group(1).decode("ascii", "replace").strip()
