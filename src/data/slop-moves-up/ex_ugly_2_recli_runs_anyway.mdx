def load_yaml_text(text, source):
    """Parse YAML text, preferring PyYAML and falling back to the built-in
    parser when PyYAML rejects the document."""
    if _pyyaml is not None:
        try:
            return _pyyaml.safe_load(text)
        except Exception:
            pass
    try:
        return _mini_yaml_parse(text)
    except ValueError as error:
        raise CliError(["Error: Cannot parse config file '%s': %s"
                        % (source, error)])
